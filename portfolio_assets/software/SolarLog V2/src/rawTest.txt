void readMuxChannelFromSerial()
{
  static char buf[8];
  static uint8_t len = 0;

  while (Serial.available())
  {
    char c = (char)Serial.read();
    if (c == '\r')
      continue; // ignore CR
    if (c == '\n')
    { // end of line
      buf[len] = '\0';
      // trim leading spaces
      char *s = buf;
      while (*s == ' ' || *s == '\t')
        ++s;

      if (s[0] >= '0' && s[0] <= '7' && s[1] == '\0')
      {
        uint8_t ch = (uint8_t)(s[0] - '0');
        MUX_Select(ch); // <-- implement this for your hardware
        Serial.print("MUX set to ");
        Serial.println(ch);
      }
      else
      {
        Serial.println("Invalid. Enter a single digit 0..7.");
      }
      len = 0; // reset buffer
      return;  // process one command per call
    }
    if (len < sizeof(buf) - 1)
      buf[len++] = c;
  }
}
