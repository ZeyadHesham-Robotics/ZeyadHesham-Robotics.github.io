void ComManager::begin(long gsmBaudRate)
{
    gsmSerial.begin(gsmBaudRate, SERIAL_8N1, GSM_RX, GSM_TX);
    delay(1000);
    modem.restart();
    Serial.println("GSM modem Initialized with baud rate: " + String(gsmBaudRate));
    while (!modem.isNetworkConnected())
    {
        Serial.print(".");
        delay(100);
    }
    Serial.println("\nNetwork connected!");
}

bool ComManager::connectWiFi()
{
    if (WiFi.status() == WL_CONNECTED)
        return true;

    Serial.printf("Connecting to WiFi SSID: %s\n", wifiSSID.c_str());
    WiFi.begin(wifiSSID.c_str(), wifiPassword.c_str());

    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 10000)
    {
        Serial.print(".");
        delay(100);
    }
    Serial.println();

    if (WiFi.status() == WL_CONNECTED)
    {
        Serial.println("WiFi connected.");
        wifiMqttClient.setServer(mqtt_server, mqtt_port);
        return true;
    }
    else
    {
        Serial.println("WiFi connection failed.");
        return false;
    }
}

bool reconnect()
{
    while (!wifiMqttClient.connected())
    {
        Serial.print("Attempting MQTT connection...");
        // Try connecting with client ID, username, and password
        if (wifiMqttClient.connect(mqtt_clientid, mqtt_user, mqtt_password))
        {
            Serial.println("connected");
            wifiMqttClient.subscribe(mqtt_topic);
            Serial.print("Subscribed to: ");
            Serial.println(mqtt_topic);
        }
        else
        {
            Serial.print("failed, rc=");
            Serial.print(wifiMqttClient.state());
            Serial.println(" retrying in 5 seconds");
            delay(5000);
        }
    }
}

// ---------- Main loop ----------
void ComManager::loop(const SensorData &data)
{
    static enum { WIFI_SEND,
                  GSM_SEND } state = WIFI_SEND;

    switch (state)
    {
    case WIFI_SEND:
        if (WiFi.status() == WL_CONNECTED)
        {
            wifiMqttClient.loop();
            if (!wifiMqttClient.connected())
            {
                if (!reconnect())
                {
                    Serial.println("WiFi MQTT failed, switching to GSM");
                    state = GSM_SEND;
                    break;
                }
            }
            publishSensorDataMQTT(wifiMqttClient, data);
            state = WIFI_SEND; // stay in WiFi mode
        }
        else
        {
            Serial.println("WiFi not available, switching to GSM");
            state = GSM_SEND;
        }
        break;

        // GSM_SEND can be re-enabled if you want fallback
    }
}

void ComManager::publishSensorDataMQTT(PubSubClient &client, const SensorData &data)
{
    // Safely format values into strings
    String soilTempStr = String(data.soilTempC, 2);
    String airTempStr = String(data.airTempC, 2);
    String airHumStr = String(data.airHumidity, 2);
    String soilPhStr = String(data.soilPH, 2);
    String soilMoistStr = String(data.soilMoisture);
    String soilECStr = String(data.soilEC);
    String nitrogenStr = String(data.nitrogen);
    String phosphorusStr = String(data.phosphorus);
    String potassiumStr = String(data.potassium);
    String battVoltStr = String(data.batteryVoltage, 2);
    String timestampStr = String(data.timestamp);
    String doorOpenStr = data.doorOpen ? "true" : "false";

    // Publish them safely
    client.publish("farm/soilTemp", soilTempStr.c_str());
    client.publish("farm/airTemp", airTempStr.c_str());
    client.publish("farm/airHumidity", airHumStr.c_str());
    client.publish("farm/soilPH", soilPhStr.c_str());
    client.publish("farm/soilMoisture", soilMoistStr.c_str());
    client.publish("farm/soilEC", soilECStr.c_str());
    client.publish("farm/nitrogen", nitrogenStr.c_str());
    client.publish("farm/phosphorus", phosphorusStr.c_str());
    client.publish("farm/potassium", potassiumStr.c_str());
    client.publish("farm/batteryVoltage", battVoltStr.c_str());
    client.publish("farm/timestamp", timestampStr.c_str());
    client.publish("farm/doorOpen", doorOpenStr.c_str());
    Serial.println("Sensor data published to MQTT.");
}
